// Fetches each code's recorded candles of 2026-11-16 and keeps 02:00-02:02Z
// in candles/<code>.json. Reads the cookie from the secret file; never prints it.
import { readFileSync, writeFileSync, existsSync, mkdirSync } from "node:fs"
mkdirSync("candles", { recursive: true })
const token = readFileSync(process.env.SESSION_TOKEN_FILE, "utf8").trim()
const codes = JSON.parse(readFileSync("codes.json")).codes.map((c) => c.code)
const from = 1794794400000, to = 1794794580000
let i = 0, failed = 0
const worker = async () => {
  while (i < codes.length) {
    const code = codes[i++]
    const file = `candles/${code}.json`
    if (existsSync(file)) continue
    const r = await fetch(`https://kumora-dashboard.kumora.workers.dev/market-live/days/2026-11-16/candles/${code}?v=5`, { headers: { Cookie: `__Secure-better-auth.session_token=${token}` } })
    if (!r.ok) { failed++; continue }
    const body = await r.json()
    writeFileSync(file, JSON.stringify(body.candles.filter((c) => c.minute >= from && c.minute < to)))
  }
}
await Promise.all(Array.from({ length: 16 }, worker))
console.log({ codes: codes.length, failed })
